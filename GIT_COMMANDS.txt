# Git команды для коммита изменений

## Проверка изменений
git status

## Добавление файлов
git add migrations/schema_sqlite.sql
git add migrations/schema_mysql.sql
git add migrations/README.md
git add migrations/QUICK_START.md
git add migrations/MIGRATION_CONSOLIDATION.md
git add migrations/CHECKLIST.md
git add migrations/legacy/
git add cmd/admin/main.go
git add MIGRATION_UPDATE.md
git add COMMIT_MESSAGE.txt

## Или добавить всё сразу
git add migrations/ cmd/admin/main.go MIGRATION_UPDATE.md COMMIT_MESSAGE.txt GIT_COMMANDS.txt

## Коммит
git commit -F COMMIT_MESSAGE.txt

## Или с кратким сообщением
git commit -m "Консолидация SQL миграций в единые файлы схемы"

## Проверка коммита
git log -1 --stat

## Push (если нужно)
git push origin main
# или
git push origin master
